{# Nota com uma casa decimal e vírgula; inteiros sem decimal (5,0 vira 5). #} {% macro nota(valor) -%} {%- set decimos = (valor * 10 + 0.5)|int -%} {%- if decimos % 10 == 0 -%}{{ decimos // 10 }}{%- else -%}{{ (decimos / 10)|string|replace('.', ',') }}{%- endif -%} {%- endmacro %} {# Barra de dificuldade estilo "HP" (modo leitura). A largura segue a média exata (.preenche-N, em passos de 2%) e a cor vai do verde (1, fácil) ao vermelho (5, difícil) conforme o nível arredondado (.nivel-N). O valor exato fica no aria-label e, com mostrar_valor, em texto ao lado; com escala, mostra a referência de 1 a 5 abaixo. Sem style inline (CSP): tudo via classes. #} {% macro barra_dificuldade(valor, mostrar_valor=true, escala=true) -%} {%- if valor is not none -%} {%- set nivel = [[(valor + 0.5)|int, 1]|max, 5]|min -%} {%- set preenche = [[((valor * 10 + 0.5)|int) * 2, 0]|max, 100]|min -%} {%- if mostrar_valor %} {{ nota(valor) }}/5 {%- endif %} {%- if escala %} {%- for n in range(1, 6) %}{{ n }}{% endfor -%} {%- endif %} {%- endif -%} {%- endmacro %}
